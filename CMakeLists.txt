cmake_minimum_required(VERSION 3.16)
project(mini_coding_agent VERSION 0.2 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(AGENT_SANITIZE "Enable address and undefined behavior sanitizers" OFF)

find_package(CURL REQUIRED)
set(CURSES_NEED_NCURSES TRUE)
if(NOT APPLE)
  set(CURSES_NEED_WIDE TRUE)
endif()
find_package(Curses REQUIRED)

add_library(agent_common src/json.c src/util.c src/protocol.c third_party/yyjson/yyjson.c)
target_include_directories(agent_common PUBLIC src)
target_include_directories(agent_common SYSTEM PUBLIC third_party)
target_compile_definitions(agent_common PUBLIC _POSIX_C_SOURCE=200809L _XOPEN_SOURCE=700 AGENT_PROJECT_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(APPLE)
  target_compile_definitions(agent_common PUBLIC _DARWIN_C_SOURCE)
endif()

add_library(agent_core src/workspace.c src/tools.c src/model_client.c src/agent.c)
target_link_libraries(agent_core PUBLIC agent_common CURL::libcurl)
add_executable(agent-server src/server.c)
target_link_libraries(agent-server PRIVATE agent_core)

add_library(agent_ui src/client_ui.c src/client_tui.c)
target_include_directories(agent_ui PRIVATE ${CURSES_INCLUDE_DIRS})
target_link_libraries(agent_ui PUBLIC agent_common PRIVATE ${CURSES_LIBRARIES})
add_executable(agent-client src/client.c)
target_link_libraries(agent-client PRIVATE agent_ui)

foreach(target agent_common agent_core agent_ui agent-server agent-client)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
if(AGENT_SANITIZE)
  target_compile_options(agent_common PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
  target_link_options(agent_common PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(agent-core-tests tests/core_tests.c)
  target_link_libraries(agent-core-tests PRIVATE agent_core)
  target_compile_options(agent-core-tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME core COMMAND agent-core-tests)
  set_tests_properties(core PROPERTIES TIMEOUT 60)
  add_executable(agent-integration-tests tests/integration_tests.c)
  target_link_libraries(agent-integration-tests PRIVATE agent_core)
  target_compile_options(agent-integration-tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME integration COMMAND agent-integration-tests $<TARGET_FILE:agent-server> $<TARGET_FILE:agent-client>)
  set_tests_properties(integration PROPERTIES TIMEOUT 90)
  add_executable(agent-ui-tests tests/ui_tests.c)
  target_link_libraries(agent-ui-tests PRIVATE agent_ui)
  target_compile_options(agent-ui-tests PRIVATE -Wall -Wextra -Wpedantic)
  find_library(UTIL_LIBRARY NAMES util)
  if(UTIL_LIBRARY)
    target_link_libraries(agent-ui-tests PRIVATE ${UTIL_LIBRARY})
  endif()
  add_test(NAME ui COMMAND agent-ui-tests $<TARGET_FILE:agent-client>)
  set_tests_properties(ui PROPERTIES TIMEOUT 60)
  add_executable(agent-live-check tests/live_check.c)
  target_link_libraries(agent-live-check PRIVATE agent_core)
  target_compile_options(agent-live-check PRIVATE -Wall -Wextra -Wpedantic)
endif()
