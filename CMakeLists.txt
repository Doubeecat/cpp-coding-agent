cmake_minimum_required(VERSION 3.16)
project(mini_coding_agent VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(CURL REQUIRED)
set(CURSES_NEED_NCURSES TRUE)
if(NOT APPLE)
  set(CURSES_NEED_WIDE TRUE)
endif()
find_package(Curses REQUIRED)
add_library(agent_core src/protocol.cpp src/model_client.cpp src/tools.cpp src/agent.cpp)
target_include_directories(agent_core PUBLIC src)
target_include_directories(agent_core SYSTEM PUBLIC third_party)
target_link_libraries(agent_core PUBLIC CURL::libcurl)
target_compile_options(agent_core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(agent_core PUBLIC AGENT_PROJECT_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(agent-server src/server.cpp)
add_executable(agent-client src/client.cpp)
add_library(agent_ui src/client_tui.cpp)
target_include_directories(agent_ui PRIVATE ${CURSES_INCLUDE_DIRS})
target_link_libraries(agent_ui PUBLIC agent_core PRIVATE ${CURSES_LIBRARIES})
target_compile_options(agent_ui PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(agent-server PRIVATE agent_core)
target_link_libraries(agent-client PRIVATE agent_ui)
include(CTest)
if(BUILD_TESTING)
  add_executable(agent-tests tests/tests.cpp)
  target_link_libraries(agent-tests PRIVATE agent_ui)
  find_library(UTIL_LIBRARY NAMES util)
  if(UTIL_LIBRARY)
    target_link_libraries(agent-tests PRIVATE ${UTIL_LIBRARY})
  endif()
  add_executable(agent-live-check tests/live_check.cpp)
  target_link_libraries(agent-live-check PRIVATE agent_core)
  add_test(NAME agent-tests COMMAND agent-tests $<TARGET_FILE:agent-server> $<TARGET_FILE:agent-client>)
  set_tests_properties(agent-tests PROPERTIES TIMEOUT 90)
endif()
